### Start a Hono app and add two parts

```sh
pnpm add @shipstack-ai/rate-limit @shipstack-ai/auth @node-rs/argon2
```

### Rate-limit the API

```ts
import { createRateLimiter, rateLimitHeaders, slidingWindow } from '@shipstack-ai/rate-limit';
import { memoryStore } from '@shipstack-ai/rate-limit/memory';

const limiter = createRateLimiter({ store: memoryStore(), algorithm: slidingWindow(10, '1m') });

app.use('/api/*', async (c, next) => {
  const result = await limiter.limit(`ip:${c.req.header('x-real-ip') ?? 'unknown'}`);
  rateLimitHeaders(result).forEach((value, name) => c.header(name, value));
  if (!result.success) return c.text('Too many requests', 429);
  await next();
});
```

</li>
<li>

### Add sign-in and a protected route

```ts
import { createAuth } from '@shipstack-ai/auth';
import { argon2Hasher } from '@shipstack-ai/auth/argon2';
import { memoryStore as authMemoryStore } from '@shipstack-ai/auth/memory';

const auth = createAuth({
  store: authMemoryStore(),
  hasher: argon2Hasher(),
  rateLimiter: limiter, // the same part guards sign-in and password reset
  requireEmailVerification: false, // until sendEmail is wired to notifications
});

app.on(['GET', 'POST'], '/auth/*', (c) => auth.handler(c.req.raw));
app.get('/api/me', async (c) => {
  const session = await auth.getSession(c.req.raw);
  return session ? c.json(session.user) : c.text('Unauthorized', 401);
});
```

</li>
<li>

### Prove it

```ts
const creds = JSON.stringify({ email: 'ann@example.com', password: 'correct horse battery staple' });
const json = { 'content-type': 'application/json' };
await app.request('/auth/sign-up', { method: 'POST', body: creds, headers: json }); // 202
const res = await app.request('/auth/sign-in', { method: 'POST', body: creds, headers: json });
const cookie = res.headers.get('set-cookie')!.split(';')[0];
(await app.request('/api/me')).status; // 401
(await app.request('/api/me', { headers: { cookie } })).status; // 200
```
